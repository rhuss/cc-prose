#!/usr/bin/env python3
"""PreToolUse hook script for intercepting tools during headless eval.

Reads tool_handlers.yaml from the workspace. Handlers contain resolved
patterns and runtime checks (from natural language `match` and `prompt`
in eval.yaml, resolved by eval-run at workspace setup time).

Supports:
- Auto-answering AskUserQuestion via per-case overrides
- Blocking tools based on env var checks (e.g., production Jira)
- Filtering Bash commands by content patterns
"""

import agent_eval._bootstrap  # noqa: F401 — auto-activate venv

import json
import os
import re
import sys
from pathlib import Path

import yaml


def main():
    input_data = json.load(sys.stdin)
    tool_name = input_data.get("tool_name", "")
    tool_input = input_data.get("tool_input", {})

    # Load handler config from workspace
    config_path = Path("tool_handlers.yaml")
    if not config_path.exists():
        sys.exit(0)

    with open(config_path) as f:
        config = yaml.safe_load(f) or {}

    # Find matching handler
    handler = _find_handler(tool_name, tool_input, config.get("handlers", []))
    if not handler:
        sys.exit(0)  # No handler — pass through

    # --- AskUserQuestion: auto-answer ---
    if tool_name == "AskUserQuestion":
        _handle_ask_user(tool_input, config, handler)
        return

    # --- Env checks: block if environment doesn't match ---
    env_checks = handler.get("env_checks", {})
    if env_checks:
        for var_name, check in env_checks.items():
            value = os.environ.get(var_name, "")
            must_contain = check.get("must_contain", [])
            if must_contain and not any(m in value.lower() for m in must_contain):
                _deny(f"Env {var_name}='{value}' doesn't match required: {must_contain}")
                return
        # All env checks passed — allow
        sys.exit(0)

    # --- Default for matched tools without specific handling: block ---
    _deny(f"Blocked by eval harness: {handler.get('match', 'matched handler')}")


def _find_handler(tool_name, tool_input, handlers):
    """Find the first handler that matches the tool call.

    Bash handlers REQUIRE input_filters — without them, a handler with
    "Bash" in patterns would silently match every Bash command and the
    default-deny in main() would block the entire skill. To prevent that
    footgun, Bash handlers without input_filters are treated as
    misconfigured: emit a stderr warning and skip (pass-through).
    Resolve them in eval-run Step 3b before relying on the handler.
    """
    for h in handlers:
        patterns = h.get("patterns", [])
        input_filters = h.get("input_filters", [])

        if tool_name == "Bash" and "Bash" in patterns:
            if not input_filters:
                print(
                    f"tool_handlers.yaml: handler {h.get('match', '?')!r} "
                    "has 'Bash' in patterns but no input_filters — "
                    "skipping (would deny all Bash). Resolve in eval-run "
                    "Step 3b.",
                    file=sys.stderr,
                )
                continue
            command = tool_input.get("command", "")
            if any(re.search(f, command, re.IGNORECASE) for f in input_filters):
                return h
            continue  # Bash matched pattern but not filter — skip

        # For other tools: match by pattern only
        for pattern in patterns:
            if pattern == tool_name:
                return h
            if pattern.endswith("*") and tool_name.startswith(pattern[:-1]):
                return h

    return None


def _handle_ask_user(tool_input, config, handler):
    """Auto-answer AskUserQuestion using case overrides, LLM, or first option.

    Resolution order for each question:
    1. Exact match in case_overrides (question text → answer)
    2. LLM-based answer (haiku) using the handler prompt + case context
    3. Fallback: pick the first option or "yes"
    """
    case_overrides = config.get("case_overrides", {})
    hook_model = config.get("hook_model")
    prompt = handler.get("prompt", "")
    if not prompt and handler.get("prompt_file"):
        try:
            prompt = Path(handler["prompt_file"]).read_text()
        except OSError:
            pass
    answers = {}
    for q in tool_input.get("questions", []):
        text = q.get("question", "")
        options = q.get("options", [])

        # 1. Exact match
        answer = case_overrides.get(text)

        # 2. LLM-based answer
        if answer is None and options:
            answer = _llm_answer(text, options, prompt, model=hook_model)

        # 3. Fallback
        if answer is None:
            answer = options[0]["label"] if options else "yes"

        answers[text] = answer

    output = {
        "hookSpecificOutput": {
            "hookEventName": "PreToolUse",
            "permissionDecision": "allow",
            "updatedInput": {
                "questions": tool_input["questions"],
                "answers": answers,
            },
        }
    }
    json.dump(output, sys.stdout)


def _llm_answer(question, options, handler_prompt, model=None):
    """Use an LLM to pick the best answer for a question.

    Reads input.yaml and answers.yaml from CWD for case-specific context.
    Returns the selected option label, or None if the API call fails.
    """
    # Load case context
    case_context = ""
    for fname in ("input.yaml", "answers.yaml"):
        p = Path(fname)
        if p.exists():
            try:
                case_context += f"\n--- {fname} ---\n{p.read_text()}\n"
            except OSError:
                pass

    option_labels = [o["label"] for o in options]
    option_list = "\n".join(
        f"  {i+1}. {o['label']}: {o.get('description', '')}"
        for i, o in enumerate(options)
    )

    prompt = f"""You are answering a question on behalf of a user during an automated evaluation run.

Handler instructions: {handler_prompt}

Case context:
{case_context}

Question: {question}

Available options:
{option_list}

Based on the handler instructions and case context, which option should be selected?
Reply with ONLY the option label text, nothing else."""

    try:
        import anthropic
        client = anthropic.Anthropic(timeout=30.0)
        response = client.messages.create(
            model=model or "claude-haiku-4-5-20251001",
            max_tokens=256,
            temperature=0,
            messages=[{"role": "user", "content": prompt}],
        )
        answer = response.content[0].text.strip()
        # Verify the answer matches an option label
        if answer in option_labels:
            print(f"LLM answered: {answer!r}", file=sys.stderr)
            return answer

Microservices split an application into small, independently deployable services that communicate over the network. Each service owns its data and runs in its own process.

The main benefit is independent deployment. Teams ship changes to one service without coordinating a release across the whole system. At Uber, this meant going from weekly monolith deploys to thousands of production changes per day, though they've since written extensively about the operational cost that came with it.

Independent scaling is real but often oversold. Most services in a typical system don't have meaningfully different load profiles. You can scale a single hot service without touching the rest, which matters when it matters, but that's fewer services than people assume.

The tradeoffs are concrete:

- Network calls replace function calls. Latency goes up. You need retries, timeouts, and circuit breakers where you previously needed nothing.
- Data consistency gets harder. Transactions that used to be a single database commit now span services. You end up building sagas or accepting eventual consistency.
- Debugging a request that touches six services requires distributed tracing. Without it, you're reading logs from six different systems and guessing at causality.
- Every service needs its own CI/CD pipeline, monitoring, and on-call rotation. The operational overhead scales with the number of services, not the number of engineers.

Microservices make sense when teams are large enough that coordination costs dominate. For a team of five, a well-structured monolith is almost always simpler. Sam Newman's "Building Microservices" puts this well: start with a monolith and extract services when you feel the pain, not before.

Install it:

config, err := pgxpool.ParseConfig("postgres://user:pass@localhost:5432/mydb")

}
config.MaxConns = 10
config.MinConns = 2

pool, err := pgxpool.NewWithConfig(context.Background(), config)

    log.Fatal(err)
}

`MaxConns` caps the number of open connections. If your application hits this limit, calls to `Acquire()` block until a connection frees up. `MinConns` keeps idle connections warm so the first few queries after a quiet period don't pay the connection setup cost.

#!/usr/bin/env python3
"""PreToolUse hook script for intercepting tools during headless eval.

Reads tool_handlers.yaml from the workspace. Handlers contain resolved
patterns and runtime checks (from natural language `match` and `prompt`
in eval.yaml, resolved by eval-run at workspace setup time).

Supports:
- Auto-answering AskUserQuestion via per-case overrides
- Blocking tools based on env var checks (e.g., production Jira)
- Filtering Bash commands by content patterns
"""

import agent_eval._bootstrap  # noqa: F401 — auto-activate venv

import json
import os
import re
import sys
from pathlib import Path

import yaml


def main():
    input_data = json.load(sys.stdin)
    tool_name = input_data.get("tool_name", "")
    tool_input = input_data.get("tool_input", {})

    # Load handler config from workspace
    config_path = Path("tool_handlers.yaml")
    if not config_path.exists():
        sys.exit(0)

    with open(config_path) as f:
        config = yaml.safe_load(f) or {}

    # Find matching handler
    handler = _find_handler(tool_name, tool_input, config.get("handlers", []))
    if not handler:
        sys.exit(0)  # No handler — pass through

    # --- AskUserQuestion: auto-answer ---
    if tool_name == "AskUserQuestion":
        _handle_ask_user(tool_input, config, handler)
        return

    # --- Env checks: block if environment doesn't match ---
    env_checks = handler.get("env_checks", {})
    if env_checks:
        for var_name, check in env_checks.items():
            value = os.environ.get(var_name, "")
            must_contain = check.get("must_contain", [])
            if must_contain and not any(m in value.lower() for m in must_contain):
                _deny(f"Env {var_name}='{value}' doesn't match required: {must_contain}")
                return
        # All env checks passed — allow
        sys.exit(0)

    # --- Default for matched tools without specific handling: block ---
    _deny(f"Blocked by eval harness: {handler.get('match', 'matched handler')}")


def _find_handler(tool_name, tool_input, handlers):
    """Find the first handler that matches the tool call.

    Bash handlers REQUIRE input_filters — without them, a handler with
    "Bash" in patterns would silently match every Bash command and the
    default-deny in main() would block the entire skill. To prevent that
    footgun, Bash handlers without input_filters are treated as
    misconfigured: emit a stderr warning and skip (pass-through).
    Resolve them in eval-run Step 3b before relying on the handler.
    """
    for h in handlers:
        patterns = h.get("patterns", [])
        input_filters = h.get("input_filters", [])

        if tool_name == "Bash" and "Bash" in patterns:
            if not input_filters:
                print(
                    f"tool_handlers.yaml: handler {h.get('match', '?')!r} "
                    "has 'Bash' in patterns but no input_filters — "
                    "skipping (would deny all Bash). Resolve in eval-run "
                    "Step 3b.",
                    file=sys.stderr,
                )
                continue
            command = tool_input.get("command", "")
            if any(re.search(f, command, re.IGNORECASE) for f in input_filters):
                return h
            continue  # Bash matched pattern but not filter — skip

        # For other tools: match by pattern only
        for pattern in patterns:
            if pattern == tool_name:
                return h
            if pattern.endswith("*") and tool_name.startswith(pattern[:-1]):
                return h

    return None


def _handle_ask_user(tool_input, config, handler):
    """Auto-answer AskUserQuestion using case overrides, LLM, or first option.

    Resolution order for each question:
    1. Exact match in case_overrides (question text → answer)
    2. LLM-based answer (haiku) using the handler prompt + case context
    3. Fallback: pick the first option or "yes"
    """
    case_overrides = config.get("case_overrides", {})
    hook_model = config.get("hook_model")
    prompt = handler.get("prompt", "")
    if not prompt and handler.get("prompt_file"):
        try:
            prompt = Path(handler["prompt_file"]).read_text()
        except OSError:
            pass
    answers = {}
    for q in tool_input.get("questions", []):
        text = q.get("question", "")
        options = q.get("options", [])

        # 1. Exact match
        answer = case_overrides.get(text)

        # 2. LLM-based answer
        if answer is None and options:
            answer = _llm_answer(text, options, prompt, model=hook_model)

        # 3. Fallback
        if answer is None:
            answer = options[0]["label"] if options else "yes"

        answers[text] = answer

    output = {
        "hookSpecificOutput": {
            "hookEventName": "PreToolUse",
            "permissionDecision": "allow",
            "updatedInput": {
                "questions": tool_input["questions"],
                "answers": answers,
            },
        }
    }
    json.dump(output, sys.stdout)


def _llm_answer(question, options, handler_prompt, model=None):
    """Use an LLM to pick the best answer for a question.

    Reads input.yaml and answers.yaml from CWD for case-specific context.
    Returns the selected option label, or None if the API call fails.
    """
    # Load case context
    case_context = ""
    for fname in ("input.yaml", "answers.yaml"):
        p = Path(fname)
        if p.exists():
            try:
                case_context += f"\n--- {fname} ---\n{p.read_text()}\n"
            except OSError:
                pass

    option_labels = [o["label"] for o in options]
    option_list = "\n".join(
        f"  {i+1}. {o['label']}: {o.get('description', '')}"
        for i, o in enumerate(options)
    )

    prompt = f"""You are answering a question on behalf of a user during an automated evaluation run.

Handler instructions: {handler_prompt}

Case context:
{case_context}

Question: {question}

Available options:
{option_list}

Based on the handler instructions and case context, which option should be selected?
Reply with ONLY the option label text, nothing else."""

    try:
        import anthropic
        client = anthropic.Anthropic(timeout=30.0)
        response = client.messages.create(
            model=model or "claude-haiku-4-5-20251001",
            max_tokens=256,
            temperature=0,
            messages=[{"role": "user", "content": prompt}],
        )
        answer = response.content[0].text.strip()
        # Verify the answer matches an option label
        if answer in option_labels:
            print(f"LLM answered: {answer!r}", file=sys.stderr)
            return answer

## REST API rate limiting

Rate limiting caps how many requests a client can make in a given time window. Without it, a single misbehaving client can starve everyone else.

### Algorithms

- Token bucket: allows short bursts above the average rate, then throttles. Most APIs use this.
- Sliding window: tracks requests across overlapping intervals for smoother enforcement.
- Fixed window: counts requests in discrete intervals. Simple to implement, but spiky at window boundaries.

### Response headers

Use these headers so clients can self-regulate:

- `X-RateLimit-Limit`: the max requests allowed per window
- `X-RateLimit-Remaining`: how many requests the client has left
- `X-RateLimit-Reset`: when the current window resets

Return `429 Too Many Requests` with a `Retry-After` header when a client exceeds their quota. Clients that respect `Retry-After` will back off automatically; clients that don't are the ones you needed rate limiting for in the first place.

def _find_handler(tool_name, tool_input, handlers):
    """Find the first handler that matches the tool call.

    Bash handlers REQUIRE input_filters — without them, a handler with
    "Bash" in patterns would silently match every Bash command and the
    default-deny in main() would block the entire skill. To prevent that
    footgun, Bash handlers without input_filters are treated as
    misconfigured: emit a stderr warning and skip (pass-through).
    Resolve them in eval-run Step 3b before relying on the handler.
    """
    for h in handlers:
        patterns = h.get("patterns", [])
        input_filters = h.get("input_filters", [])

        if tool_name == "Bash" and "Bash" in patterns:
            if not input_filters:
                print(
                    f"tool_handlers.yaml: handler {h.get('match', '?')!r} "
                    "has 'Bash' in patterns but no input_filters — "
                    "skipping (would deny all Bash). Resolve in eval-run "
                    "Step 3b.",
                    file=sys.stderr,
                )
                continue
            command = tool_input.get("command", "")
            if any(re.search(f, command, re.IGNORECASE) for f in input_filters):
                return h
            continue  # Bash matched pattern but not filter — skip

        # For other tools: match by pattern only
        for pattern in patterns:
            if pattern == tool_name:
                return h
            if pattern.endswith("*") and tool_name.startswith(pattern[:-1]):
                return h

    return None

    case_context = ""

            print(f"LLM answered: {answer!r}", file=sys.stderr)

The WebAssembly specification reached version 2.0 in 2024, adding features like multiple return values and reference types. All four major browsers ship full Wasm 2.0 support, and server-side runtimes like Wasmtime and WasmEdge have kept pace.

Wasm modules run fast. A 2023 benchmark by the University of Stuttgart measured Wasm at 1.1-1.5x native speed for compute-heavy workloads (matrix math, image processing), though I/O-bound tasks showed less improvement. Production adoption has grown too. Fermyon reported over 10,000 Wasm-based microservices running on their platform by mid-2024, up from a few hundred the year before.

The sandbox model is the interesting part. Each Wasm module runs in its own linear memory with no direct access to the host OS. That is a real isolation win compared to shared-process plugin architectures. The Component Model proposal takes this further by defining typed interfaces between modules. No major vulnerabilities have hit the core spec so far, though side-channel attacks against the JIT tier remain an active research area.

WebAssembly is not a JavaScript replacement. JavaScript handles DOM manipulation, event wiring, and the glue code that holds web apps together. Wasm handles the parts where you need predictable performance: codecs, physics engines, cryptography, data processing. They share the same runtime and call each other through well-defined bindings.

Tooling depends on your source language. C, C++, and Rust compile to Wasm through LLVM and produce small, optimized binaries. Go added native Wasm support but produces larger output. Languages like Python and Ruby can run in Wasm through interpreters (Pyodide, ruby.wasm), though the cold-start cost is noticeable. The toolchain is functional, not polished. Expect rough edges in debugging and profiling.
